% Pseudocode in the style of Cormen, Leiserson, Rivest and Stein: one
% statement per numbered line, procedure names in small capitals, and a
% wider gap after the last line of each group of lines that belong together
% (end such a line with \groupend instead of \\).
\newcounter{codeline}
% Procedure names in bold capitals: each hyphen-separated word keeps its first
% letter at full height and sets the rest as reduced capitals, as in
% \proc{Matrix-Multiply}.
\ExplSyntaxOn
\seq_new:N \l__proc_words_seq
\NewDocumentCommand{\proc}{m}{
  \textsf{\bfseries
    \seq_set_split:Nnn \l__proc_words_seq { - } { #1 }
    \seq_map_indexed_inline:Nn \l__proc_words_seq
      {
        \int_compare:nT { ##1 > 1 } { - }
        \tl_head:n { ##2 }
        \scalebox{0.8}{ \text_uppercase:n { \tl_tail:n { ##2 } } }
      }
  }
}
\ExplSyntaxOff
% Arguments: the procedure's signature, a description of its inputs, and a
% description of what it returns or computes.
\newenvironment{procedure}[3]{%
  \setcounter{codeline}{0}%
  \begin{tcolorbox}[enhanced,colback=codefill,colframe=codefill,boxrule=0pt,arc=4pt,left=10pt,right=8pt,top=6pt,bottom=4pt]%
  \small\linespread{1.18}\selectfont\noindent#1\par\vspace{3pt}%
  {\footnotesize\color{muted}\noindent\textit{Input:} #2\par\noindent\textit{Output:} #3\par}\vspace{4pt}%
  \begin{tabbing}%
  \hspace*{1.6em}\=\hspace{1.2em}\=\hspace{1.2em}\=\hspace{1.2em}\=\hspace{1.2em}\=\kill}{%
  \end{tabbing}\end{tcolorbox}}
